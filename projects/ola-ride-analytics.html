SQL reporting views
CREATE DATABASE Ola;
USE Ola;

-- Source table: Ola_Bengaluru_Ride_Data

CREATE VIEW SuccessfulBookings AS
SELECT * FROM Ola_Bengaluru_Ride_Data
WHERE Booking_Status = 'Success';

CREATE VIEW Avg_RideDistance_By_Vehicle AS
SELECT Vehicle_Type, ROUND(AVG(Ride_Distance), 2) AS Avg_Ride_Distance
FROM Ola_Bengaluru_Ride_Data
GROUP BY Vehicle_Type;

CREATE VIEW CustomerCancelled_Rides_Count AS
SELECT COUNT(*) AS No_cancelled_by_customers
FROM Ola_Bengaluru_Ride_Data
WHERE Cancelled_Rides_by_Customer = 1;

CREATE VIEW Top_5Customers AS
SELECT TOP 5 Customer_ID, COUNT(Booking_ID) AS No_rides
FROM Ola_Bengaluru_Ride_Data
GROUP BY Customer_ID
ORDER BY No_rides DESC;

CREATE VIEW DriverCancelled_Personal_Car_Issues AS
SELECT COUNT(*) AS No_cancelled_rides
FROM Ola_Bengaluru_Ride_Data
WHERE Reason_for_cancelling_by_Driver = 'Personal & Car related issues';

CREATE VIEW PrimeSedan_Ratings AS
SELECT Vehicle_Type, MAX(Customer_Rating) AS Max_Rating,
       MIN(Customer_Rating) AS Min_Rating
FROM Ola_Bengaluru_Ride_Data
WHERE Vehicle_Type = 'Prime Sedan'
GROUP BY Vehicle_Type;

CREATE VIEW UPIPayment AS
SELECT * FROM Ola_Bengaluru_Ride_Data
WHERE Payment_Method = 'UPI';

CREATE VIEW AvgCustomer_Rating_By_Vehicle AS
SELECT Vehicle_Type, AVG(Customer_Rating) AS Avg_Rating
FROM Ola_Bengaluru_Ride_Data
GROUP BY Vehicle_Type;

CREATE VIEW TotalSuccessful_Booking_Value AS
SELECT SUM(Booking_Value) AS Total_Value
FROM Ola_Bengaluru_Ride_Data
WHERE Booking_Status = 'Success';

CREATE VIEW IncompleteRides_Details AS
SELECT Booking_ID, Incomplete_Rides
FROM Ola_Bengaluru_Ride_Data
WHERE Incomplete_Rides = 1;
